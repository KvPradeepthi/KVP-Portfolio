// Batch record aggregation & audit notifications
export async function dispatchLatecomerReports() {
  const cutoff = new Date();
  cutoff.setHours(9, 15, 0);

  const entries = await AttendanceModel.aggregate([
    { $match: { scanTime: { $gt: cutoff } } },
    { $group: { _id: "$department", count: { $sum: 1 } } }
  ]);

  await NotificationQueue.publish("daily.audit", entries);
  return { status: "200_DISPATCHED", count: entries.length };
}
